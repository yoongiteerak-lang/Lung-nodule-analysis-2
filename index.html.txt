    <script>
        let rawImageObj = new Image();
        let isImageLoaded = false;
        let rawPixelData = null;      
        let enhancedPixelData = null; 
        const MAX_WIDTH = 600;

        let currentMode = 'DRAW'; 
        let isDrawing = false;
        let roiPoints = [];
        let roiMask = null;
        let seedPoints = []; 

        const imageUpload = document.getElementById('imageUpload');
        const canvasOriginal = document.getElementById('canvasOriginal');
        const canvasResult = document.getElementById('canvasResult');
        const ctxOriginal = canvasOriginal.getContext('2d', { willReadFrequently: true });
        const ctxResult = canvasResult.getContext('2d');
        const sourceContainer = document.getElementById('sourceContainer');
        const markersContainer = document.getElementById('markersContainer');

        const brightnessSlider = document.getElementById('brightness');
        const contrastSlider = document.getElementById('contrast');
        const resetEnhanceBtn = document.getElementById('resetEnhanceBtn');
        
        const toleranceSlider = document.getElementById('tolerance');
        const erosionSlider = document.getElementById('erosion');
        const dilationSlider = document.getElementById('dilation');
        const showOverlayCb = document.getElementById('showOverlay');
        const resetDrawBtn = document.getElementById('resetDrawBtn');
        const clearSeedsBtn = document.getElementById('clearSeedsBtn');
        const step1UI = document.getElementById('step1');
        const step2UI = document.getElementById('step2');

        const emptyAnalysis = document.getElementById('emptyAnalysis');
        const analysisResult = document.getElementById('analysisResult');

        brightnessSlider.addEventListener('input', (e) => { document.getElementById('brightnessVal').innerText = e.target.value; applyImageEnhancement(); });
        contrastSlider.addEventListener('input', (e) => { document.getElementById('contrastVal').innerText = e.target.value; applyImageEnhancement(); });
        resetEnhanceBtn.addEventListener('click', () => { 
            brightnessSlider.value = 0; contrastSlider.value = 0; 
            document.getElementById('brightnessVal').innerText = "0"; document.getElementById('contrastVal').innerText = "0";
            applyImageEnhancement(); 
        });

        toleranceSlider.addEventListener('input', (e) => { document.getElementById('toleranceVal').innerText = e.target.value; processSegmentation(); });
        erosionSlider.addEventListener('input', (e) => { document.getElementById('erosionVal').innerText = e.target.value; processSegmentation(); });
        dilationSlider.addEventListener('input', (e) => { document.getElementById('dilationVal').innerText = e.target.value; processSegmentation(); });
        showOverlayCb.addEventListener('change', processSegmentation);
        
        resetDrawBtn.addEventListener('click', resetROI);
        clearSeedsBtn.addEventListener('click', () => {
            seedPoints = [];
            clearMarkers();
            clearSeedsBtn.classList.add('hidden');
            hideAnalysis();
            redrawOriginalCanvas();
            clearResultCanvas();
        });

        imageUpload.addEventListener('change', function(e) {
            const file = e.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = function(event) {
                rawImageObj.onload = function() {
                    document.getElementById('uploadPrompt').style.display = 'none';
                    let width = rawImageObj.width;
                    let height = rawImageObj.height;
                    if (width > MAX_WIDTH) { height = Math.floor(height * (MAX_WIDTH / width)); width = MAX_WIDTH; }

                    canvasOriginal.width = width; canvasOriginal.height = height;
                    canvasResult.width = width; canvasResult.height = height;

                    ctxOriginal.drawImage(rawImageObj, 0, 0, width, height);
                    rawPixelData = ctxOriginal.getImageData(0, 0, width, height);
                    isImageLoaded = true;
                    resetEnhanceBtn.click(); 
                }
                rawImageObj.src = event.target.result;
            }
            reader.readAsDataURL(file);
        });

        function applyImageEnhancement() {
            if (!isImageLoaded || !rawPixelData) return;
            const width = canvasOriginal.width;
            const height = canvasOriginal.height;
            const b = parseInt(brightnessSlider.value);
            const c = parseInt(contrastSlider.value);

            enhancedPixelData = new ImageData(new Uint8ClampedArray(rawPixelData.data), width, height);
            const data = enhancedPixelData.data;
            const factor = (259 * (c + 255)) / (255 * (259 - c));

            for (let i = 0; i < data.length; i += 4) {
                data[i] = clamp(factor * (data[i] - 128) + 128 + b);       
                data[i+1] = clamp(factor * (data[i+1] - 128) + 128 + b);   
                data[i+2] = clamp(factor * (data[i+2] - 128) + 128 + b);   
            }

            redrawOriginalCanvas();
            if (seedPoints.length > 0) processSegmentation();
        }
        function clamp(value) { return Math.max(0, Math.min(255, value)); }

        function getPointerPos(clientX, clientY) {
            const rect = canvasOriginal.getBoundingClientRect();
            const scaleX = canvasOriginal.width / rect.width;
            const scaleY = canvasOriginal.height / rect.height;
            return {
                x: Math.floor((clientX - rect.left) * scaleX),
                y: Math.floor((clientY - rect.top) * scaleY),
                rawX: clientX, rawY: clientY
            };
        }

        function handlePointerDown(e) {
            if (!isImageLoaded) return;
            if (e.type === 'touchstart') e.preventDefault(); 
            
            let clientX = e.type === 'touchstart' ? e.touches[0].clientX : e.clientX;
            let clientY = e.type === 'touchstart' ? e.touches[0].clientY : e.clientY;
            const pos = getPointerPos(clientX, clientY);

            if (currentMode === 'DRAW') {
                isDrawing = true;
                roiPoints = [{ x: pos.x, y: pos.y }];
                redrawOriginalCanvas();
            } else if (currentMode === 'SEED') {
                if (roiMask && roiMask[pos.y * canvasOriginal.width + pos.x] === 0) return; 

                seedPoints.push({ x: pos.x, y: pos.y });
                
                const markerX = pos.rawX - sourceContainer.getBoundingClientRect().left;
                const markerY = pos.rawY - sourceContainer.getBoundingClientRect().top;
                addMarkerVisual(markerX, markerY);

                clearSeedsBtn.classList.remove('hidden'); 
                processSegmentation();
            }
        }

        function handlePointerMove(e) {
            if (!isDrawing || currentMode !== 'DRAW') return;
            if (e.type === 'touchmove') e.preventDefault();
            let clientX = e.type === 'touchmove' ? e.touches[0].clientX : e.clientX;
            let clientY = e.type === 'touchmove' ? e.touches[0].clientY : e.clientY;
            const pos = getPointerPos(clientX, clientY);
            roiPoints.push({ x: pos.x, y: pos.y });
            redrawOriginalCanvas();
        }

        function handlePointerUp(e) {
            if (!isDrawing || currentMode !== 'DRAW') return;
            if (e.type === 'touchend') e.preventDefault();
            isDrawing = false;
            if (roiPoints.length > 5) {
                roiPoints.push({ x: roiPoints[0].x, y: roiPoints[0].y });
                generateROIMask();
                currentMode = 'SEED';
                updateStepUI();
                redrawOriginalCanvas();
            } else {
                roiPoints = [];
                redrawOriginalCanvas();
            }
        }

        canvasOriginal.addEventListener('mousedown', handlePointerDown);
        canvasOriginal.addEventListener('mousemove', handlePointerMove);
        canvasOriginal.addEventListener('mouseup', handlePointerUp);
        canvasOriginal.addEventListener('mouseleave', () => { if (isDrawing) { isDrawing = false; roiPoints = []; redrawOriginalCanvas(); }});
        canvasOriginal.addEventListener('touchstart', handlePointerDown, { passive: false });
        canvasOriginal.addEventListener('touchmove', handlePointerMove, { passive: false });
        canvasOriginal.addEventListener('touchend', handlePointerUp, { passive: false });

        function addMarkerVisual(x, y) {
            const marker = document.createElement('div');
            marker.className = 'seed-marker';
            marker.style.left = x + 'px';
            marker.style.top = y + 'px';
            marker.style.display = 'block';
            markersContainer.appendChild(marker);
        }
        function clearMarkers() {
            markersContainer.innerHTML = '';
        }

        function resetROI() {
            roiPoints = []; roiMask = null; seedPoints = [];
            currentMode = 'DRAW'; clearMarkers(); clearSeedsBtn.classList.add('hidden');
            updateStepUI(); hideAnalysis();
            if(isImageLoaded) { redrawOriginalCanvas(); clearResultCanvas(); }
        }

        function updateStepUI() {
            if (currentMode === 'DRAW') {
                step1UI.className = "step-active border-l-4 p-2 rounded-r text-xs font-medium transition-colors";
                step2UI.className = "step-inactive border-l-4 p-2 rounded-r text-xs font-medium transition-colors";
                canvasOriginal.style.cursor = "crosshair";
            } else {
                step1UI.className = "step-inactive border-l-4 p-2 rounded-r text-xs font-medium transition-colors";
                step2UI.className = "step-active border-l-4 p-2 rounded-r text-xs font-medium transition-colors";
                canvasOriginal.style.cursor = "crosshair"; 
            }
        }

        function redrawOriginalCanvas() {
            if (!isImageLoaded || !enhancedPixelData) return;
            ctxOriginal.putImageData(enhancedPixelData, 0, 0);

            if (roiPoints.length > 0) {
                ctxOriginal.beginPath();
                ctxOriginal.moveTo(roiPoints[0].x, roiPoints[0].y);
                for (let i = 1; i < roiPoints.length; i++) ctxOriginal.lineTo(roiPoints[i].x, roiPoints[i].y);

                if (currentMode === 'SEED') {
                    ctxOriginal.closePath();
                    ctxOriginal.fillStyle = 'rgba(99, 102, 241, 0.15)'; 
                    ctxOriginal.fill();
                    ctxOriginal.strokeStyle = '#6366f1';
                    ctxOriginal.setLineDash([5, 5]);
                    ctxOriginal.lineWidth = 1.5;
                } else {
                    ctxOriginal.strokeStyle = '#10b981'; 
                    ctxOriginal.setLineDash([]);
                    ctxOriginal.lineWidth = 2.5;
                }
                ctxOriginal.stroke();
                ctxOriginal.setLineDash([]);
            }
        }

        function generateROIMask() {
            const w = canvasOriginal.width, h = canvasOriginal.height;
            const offC = document.createElement('canvas'); offC.width = w; offC.height = h;
            const offCtx = offC.getContext('2d');
            offCtx.fillStyle = '#000'; offCtx.fillRect(0, 0, w, h);
            offCtx.beginPath(); offCtx.moveTo(roiPoints[0].x, roiPoints[0].y);
            for (let i = 1; i < roiPoints.length; i++) offCtx.lineTo(roiPoints[i].x, roiPoints[i].y);
            offCtx.closePath(); offCtx.fillStyle = '#fff'; offCtx.fill();
            
            const maskData = offCtx.getImageData(0, 0, w, h).data;
            roiMask = new Uint8Array(w * h);
            for (let i = 0; i < w * h; i++) roiMask[i] = maskData[i * 4] > 128 ? 1 : 0;
        }

        function clearResultCanvas() {
            ctxResult.clearRect(0, 0, canvasResult.width, canvasResult.height);
            if(enhancedPixelData) ctxResult.putImageData(enhancedPixelData, 0, 0);
        }

        function processSegmentation() {
            if (!isImageLoaded || seedPoints.length === 0 || !enhancedPixelData || !roiMask) return;

            const w = canvasOriginal.width, h = canvasOriginal.height;
            const data = enhancedPixelData.data;
            const tolerance = parseInt(toleranceSlider.value);
            
            let mask = new Uint8Array(w * h);
            let stack = [];

            let seedIntensities = [];
            seedPoints.forEach(p => {
                const sIdx = (p.y * w + p.x) * 4;
                const gray = 0.299*data[sIdx] + 0.587*data[sIdx+1] + 0.114*data[sIdx+2];
                seedIntensities.push(gray);
                
                mask[p.y * w + p.x] = 1;
                stack.push(p.x, p.y);
            });

            const dx = [0, 0, -1, 1], dy = [-1, 1, 0, 0];

            while (stack.length > 0) {
                const cy = stack.pop(), cx = stack.pop();
                for (let i = 0; i < 4; i++) {
                    const nx = cx + dx[i], ny = cy + dy[i];
                    if (nx >= 0 && nx < w && ny >= 0 && ny < h) {
                        if (roiMask[ny * w + nx] === 0) continue; 
                        
                        const nIdx = ny * w + nx;
                        if (mask[nIdx] === 0) {
                            const pIdx = nIdx * 4;
                            const nGray = 0.299*data[pIdx] + 0.587*data[pIdx+1] + 0.114*data[pIdx+2];
                            
                            let matchFound = false;
                            for(let j=0; j<seedIntensities.length; j++) {
                                if (Math.abs(nGray - seedIntensities[j]) <= tolerance) {
                                    matchFound = true;
                                    break;
                                }
                            }

                            if (matchFound) {
                                mask[nIdx] = 1; 
                                stack.push(nx, ny);
                            }
                        }
                    }
                }
            }

            const eIter = parseInt(erosionSlider.value), dIter = parseInt(dilationSlider.value);
            for (let i = 0; i < eIter; i++) mask = morphErode(mask, w, h);
            for (let i = 0; i < dIter; i++) mask = morphDilate(mask, w, h);

            drawMaskResult(mask, w, h);
            extractAdvancedRadiomics(mask, w, h, data);
        }

        function drawMaskResult(mask, w, h) {
            const data = enhancedPixelData.data;
            const overlay = showOverlayCb.checked;
            const resImg = ctxResult.createImageData(w, h);
            const rData = resImg.data;

            for (let i = 0; i < mask.length; i++) {
                const p = i * 4;
                if (overlay) {
                    if (mask[i] === 1) {
                        rData[p] = Math.min(255, data[p] + 130); 
                        rData[p+1] = data[p+1] * 0.3;
                        rData[p+2] = data[p+2] * 0.3;
                    } else {
                        rData[p] = data[p]; rData[p+1] = data[p+1]; rData[p+2] = data[p+2];
                    }
                } else {
                    const v = mask[i] === 1 ? 255 : 0;
                    rData[p] = v; rData[p+1] = v; rData[p+2] = v;
                }
                rData[p+3] = 255;
            }
            ctxResult.putImageData(resImg, 0, 0);
        }

        function morphErode(src, w, h) {
            let dst = new Uint8Array(src.length);
            for (let y = 1; y < h-1; y++) for (let x = 1; x < w-1; x++) {
                let m = 1;
                for (let dy = -1; dy <= 1; dy++) for (let dx = -1; dx <= 1; dx++) {
                    if (src[(y+dy)*w + (x+dx)] === 0) m = 0;
                }
                dst[y*w+x] = m;
            }
            return dst;
        }
        function morphDilate(src, w, h) {
            let dst = new Uint8Array(src.length);
            for (let y = 1; y < h-1; y++) for (let x = 1; x < w-1; x++) {
                let m = 0;
                for (let dy = -1; dy <= 1; dy++) for (let dx = -1; dx <= 1; dx++) {
                    if (src[(y+dy)*w + (x+dx)] === 1) m = 1;
                }
                dst[y*w+x] = m;
            }
            return dst;
        }

        function extractAdvancedRadiomics(mask, w, h, pixelData) {
            let area = 0, perimeter = 0;
            let intensities = [];

            for (let y = 0; y < h; y++) {
                for (let x = 0; x < w; x++) {
                    let idx = y * w + x;
                    if (mask[idx] === 1) {
                        area++;
                        if (x===0 || x===w-1 || y===0 || y===h-1 || 
                            mask[y*w+(x-1)]===0 || mask[y*w+(x+1)]===0 || mask[(y-1)*w+x]===0 || mask[(y+1)*w+x]===0) {
                            perimeter++;
                        }
                        const pIdx = idx * 4;
                        const gray = 0.299*pixelData[pIdx] + 0.587*pixelData[pIdx+1] + 0.114*pixelData[pIdx+2];
                        intensities.push(gray);
                    }
                }
            }

            if (area < 10) { hideAnalysis(); return; } 

            let circularity = (perimeter > 0) ? Math.min(1.0, (4 * Math.PI * area) / (perimeter * perimeter)) : 0;
            let mean = intensities.reduce((a, b) => a + b, 0) / area;
            let variance = intensities.reduce((a, b) => a + Math.pow(b - mean, 2), 0) / area;
            let stdDev = Math.sqrt(variance);

            let z = -4.5; 
            z += (area / 1000) * 1.8; 
            z += (1.0 - circularity) * 3.5;
            z += (stdDev / 30) * 1.5;

            let probability = 1 / (1 + Math.exp(-z));
            let percentage = (probability * 100).toFixed(1);

            updateRadiomicsUI(area, circularity, mean, stdDev, percentage);
        }

        function updateRadiomicsUI(area, circ, mean, std, probPercent) {
            emptyAnalysis.style.display = 'none';
            analysisResult.style.display = 'flex';

            document.getElementById('featArea').innerText = area.toLocaleString();
            document.getElementById('featCirc').innerText = circ.toFixed(3);
            document.getElementById('featMean').innerText = mean.toFixed(1);
            document.getElementById('featStd').innerText = std.toFixed(2);

            const numProb = parseFloat(probPercent);
            const banner = document.getElementById('riskBanner');
            const bgProgress = document.getElementById('riskBgProgress');
            const pctLabel = document.getElementById('riskPercentage');
            const conclusion = document.getElementById('riskConclusion');

            pctLabel.innerText = probPercent + "%";
            setTimeout(() => { bgProgress.style.width = probPercent + "%"; }, 100);

            if (numProb >= 65) {
                banner.className = "p-5 rounded-xl mb-5 text-center shadow-lg border relative overflow-hidden bg-slate-800 border-red-900 text-red-400";
                bgProgress.className = "absolute top-0 left-0 h-full transition-all duration-1000 ease-out bg-red-600/20";
                conclusion.innerHTML = "<i class='fas fa-exclamation-triangle'></i> High Risk (Malignant Suspected)";
            } else if (numProb >= 30) {
                banner.className = "p-5 rounded-xl mb-5 text-center shadow-lg border relative overflow-hidden bg-slate-800 border-amber-900 text-amber-400";
                bgProgress.className = "absolute top-0 left-0 h-full transition-all duration-1000 ease-out bg-amber-600/20";
                conclusion.innerHTML = "<i class='fas fa-search'></i> Intermediate Risk (Indeterminate)";
            } else {
                banner.className = "p-5 rounded-xl mb-5 text-center shadow-lg border relative overflow-hidden bg-slate-800 border-emerald-900 text-emerald-400";
                bgProgress.className = "absolute top-0 left-0 h-full transition-all duration-1000 ease-out bg-emerald-600/20";
                conclusion.innerHTML = "<i class='fas fa-check-circle'></i> Low Risk (Likely Benign)";
            }
        }

        function hideAnalysis() {
            emptyAnalysis.style.display = 'flex';
            analysisResult.style.display = 'none';
        }
